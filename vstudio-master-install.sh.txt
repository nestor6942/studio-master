import os
import tempfile
from fastapi import FastAPI, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware
from faster_whisper import WhisperModel

app = FastAPI(title="V-Studio AI Auto-Captions Service")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

model_size = os.getenv("WHISPER_MODEL", "base")
compute_type = os.getenv("COMPUTE_TYPE", "int8")
model = WhisperModel(model_size, device="cpu", compute_type=compute_type)

@app.post("/transcribe")
async def transcribe_word_level(file: UploadFile = File(...)):
    with tempfile.NamedTemporaryFile(delete=False, suffix=".wav") as tmp:
        content = await file.read()
        tmp.write(content)
        tmp_path = tmp.name

    try:
        segments, info = model.transcribe(tmp_path, word_timestamps=True, language="es")
        phrases = []
        for idx, segment in enumerate(segments):
            words = []
            if segment.words:
                for w in segment.words:
                    words.append({
                        "text": w.word.strip(),
                        "start": round(w.start, 2),
                        "end": round(w.end, 2),
                        "highlightColor": 16770560
                    })
            phrases.append({
                "id": f"phrase_{idx}",
                "startTime": round(segment.start, 2),
                "endTime": round(segment.end, 2),
                "words": words
            })
        return {"language": info.language, "phrases": phrases}
    finally:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)

import * as PIXI from 'pixi.js';

export interface WordTimestamp {
  text: string;
  start: number;
  end: number;
  highlightColor?: number;
}

export interface CaptionPhrase {
  id: string;
  startTime: number;
  endTime: number;
  words: WordTimestamp[];
}

export class DynamicCaptionRenderer {
  public container: PIXI.Container;
  private wordSprites: PIXI.Text[] = [];
  private highlightBox: PIXI.Graphics;
  private activePhraseId: string | null = null;

  constructor() {
    this.container = new PIXI.Container();
    this.highlightBox = new PIXI.Graphics();
    this.container.addChild(this.highlightBox);
  }

  public buildPhraseLayout(phrase: CaptionPhrase, canvasWidth: number, canvasHeight: number): void {
    if (this.activePhraseId === phrase.id) return;
    this.activePhraseId = phrase.id;

    this.wordSprites.forEach((s) => s.destroy());
    this.wordSprites = [];

    const spacing = 20;
    let totalWidth = 0;

    for (const wordData of phrase.words) {
      const style = new PIXI.TextStyle({
        fontFamily: 'Arial Black, Impact, sans-serif',
        fontSize: 64,
        fontWeight: '900',
        fill: '#FFFFFF',
        stroke: { color: '#000000', width: 12, join: 'round' },
      });

      const textObj = new PIXI.Text({ text: wordData.text.toUpperCase(), style });
      textObj.anchor.set(0.5, 0.5);
      this.container.addChild(textObj);
      this.wordSprites.push(textObj);
      totalWidth += textObj.width + spacing;
    }

    totalWidth -= spacing;
    let currentX = (canvasWidth - totalWidth) / 2;
    const baseY = canvasHeight * 0.72;

    for (const sprite of this.wordSprites) {
      sprite.x = currentX + sprite.width / 2;
      sprite.y = baseY;
      currentX += sprite.width + spacing;
    }
  }

  public updateAtTime(currentTime: number, phrase: CaptionPhrase): void {
    this.highlightBox.clear();

    phrase.words.forEach((word, index) => {
      const sprite = this.wordSprites[index];
      if (!sprite) return;

      const isSpokenNow = currentTime >= word.start && currentTime <= word.end;

      if (isSpokenNow) {
        const duration = Math.max(0.05, word.end - word.start);
        const progress = (currentTime - word.start) / duration;
        const popScale = progress < 0.25 ? 1.22 : 1.15;

        sprite.scale.set(popScale);
        sprite.style.fill = 0xFFE600;

        this.highlightBox
          .roundRect(
            sprite.x - (sprite.width * popScale) / 2 - 16,
            sprite.y - (sprite.height * popScale) / 2 - 8,
            sprite.width * popScale + 32,
            sprite.height * popScale + 16,
            16
          )
          .fill({ color: 0x7C3AED, alpha: 0.95 });
      } else {
        sprite.scale.set(1.0);
        sprite.style.fill = currentTime > word.end ? 0xCBD5E1 : 0xFFFFFF;
      }
    });
  }
}

    this.canvasHeight = height;

      height,

      '1:1':  { w: 1080, h: 1080 },

    }
  }

      this.captionRenderer.buildPhraseLayout(activePhrase, this.canvasWidth, this.canvasHeight);
      this.captionRenderer.updateAtTime(this.currentTime, activePhrase);
    }
  }
}

  const [isPublishing, setIsPublishing] = useState(false);
  const [publishResults, setPublishResults] = useState<PlatformPublishResult[] | null>(null);

  const {
    projectTitle,
    aspectRatio,
    isPlaying,
    currentTime,
    totalDuration,
    zoomLevel,
    tracks,
    clips,
    setEngineInstance,
    setAspectRatio,
    togglePlay,
    seekTo,
  } = useEditorStore();

  useEffect(() => {
    if (!canvasContainerRef.current) return;
    const engine = new HybridMediaEngine();
    engine.init(canvasContainerRef.current, 1080, 1920).then(() => {
      setEngineInstance(engine);
    });
  }, [setEngineInstance]);

  useEffect(() => {
    if (!isPlaying) return;
    let rafId: number;
    let last = performance.now();

    const loop = (now: number) => {
      const dt = (now - last) / 1000;
      last = now;
      const next = useEditorStore.getState().currentTime + dt;
      if (next >= totalDuration) {
        useEditorStore.getState().seekTo(0);
        useEditorStore.getState().togglePlay();
      } else {
        useEditorStore.getState().seekTo(next);
        rafId = requestAnimationFrame(loop);
      }
    };

    rafId = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(rafId);
  }, [isPlaying, totalDuration]);

  const togglePlatform = (platform: SocialPlatform) => {
    setSelectedPlatforms((prev) =>
      prev.includes(platform) ? prev.filter((p) => p !== platform) : [...prev, platform]
    );
  };

  const handlePublishToSocials = async () => {
    if (selectedPlatforms.length === 0) return;
    setIsPublishing(true);
    try {
      const res = await fetch('/api/social/publish', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          projectId: 'proj_viral_001',
          title: projectTitle,
          caption: postCaption,
          hashtags: ['viral', 'reels', 'tiktok', 'shorts'],
          platforms: selectedPlatforms,
        }),
      });
      const data = await res.json();
      if (data.results) setPublishResults(data.results);
    } finally {
      setIsPublishing(false);
    }
  };

  return (
    <div className="flex flex-col h-screen w-screen bg-zinc-950 text-zinc-100 select-none overflow-hidden">
      <header className="h-14 border-b border-zinc-800 bg-zinc-900 px-4 flex items-center justify-between">
        <div className="flex items-center gap-3">
          <span className="h-3 w-3 rounded-full bg-violet-500 animate-pulse" />
          <span className="font-bold text-white">V-STUDIO PRO</span>
          <span className="text-zinc-600">|</span>
          <span className="text-sm text-zinc-300">{projectTitle}</span>
        </div>

        <div className="flex items-center gap-1 bg-zinc-950 p-1 rounded-lg border border-zinc-800">
          {(['9:16', '1:1', '16:9'] as AspectRatioPreset[]).map((ratio) => (
            <button
              key={ratio}
              onClick={() => setAspectRatio(ratio)}
              className={`px-3 py-1 text-xs font-semibold rounded ${
                aspectRatio === ratio ? 'bg-violet-600 text-white' : 'text-zinc-400 hover:text-white'
              }`}
            >
              {ratio}
            </button>
          ))}
        </div>

        <div className="flex items-center gap-3">
          <button
            onClick={() => setShowSafeZones(!showSafeZones)}
            className="px-3 py-1.5 text-xs rounded border border-emerald-500/40 bg-emerald-500/10 text-emerald-300"
          >
            {showSafeZones ? '👁 Safe Zone: ON' : '👁 Safe Zone: OFF'}
          </button>
          <button
            onClick={() => {
              setPublishResults(null);
              setIsPublishModalOpen(true);
            }}
            className="px-4 py-1.5 text-xs font-bold bg-gradient-to-r from-violet-600 to-fuchsia-600 text-white rounded-md shadow-lg"
          >
            🚀 Exportar y Publicar en Redes
          </button>
        </div>
      </header>

      <div className="flex flex-1 min-h-0">
        <main className="flex-1 bg-zinc-950 flex items-center justify-center p-6">
          <div
            className="relative shadow-2xl border border-zinc-800 bg-black overflow-hidden"
            style={{
              height: '100%',
              maxHeight: '430px',
              aspectRatio: aspectRatio === '9:16' ? '9/16' : aspectRatio === '1:1' ? '1/1' : '16/9',
            }}
          >
            <div ref={canvasContainerRef} className="w-full h-full" />
            {showSafeZones && aspectRatio === '9:16' && (
              <div className="absolute inset-0 pointer-events-none border border-emerald-500/30">
                <div className="absolute top-0 inset-x-0 h-[12%] bg-rose-500/15 border-b border-dashed border-rose-400/40 flex items-center justify-center text-[10px] text-rose-300">
                  Zona Superior TikTok / Reels
                </div>
              </div>
            )}
          </div>
        </main>
      </div>

      <div className="h-11 bg-zinc-900 border-y border-zinc-800 px-4 flex items-center justify-between">
        <div className="flex items-center gap-4">
          <button
            onClick={togglePlay}
            className="h-8 w-8 rounded-full bg-violet-600 text-white font-bold flex items-center justify-center"
          >
            {isPlaying ? '⏸' : '▶'}
          </button>
          <span className="font-mono text-xs text-violet-400">
            {currentTime.toFixed(2)}s / {totalDuration.toFixed(2)}s
          </span>
        </div>
      </div>

      <section className="h-44 bg-zinc-950 flex overflow-hidden">
        <div className="w-48 bg-zinc-900 border-r border-zinc-800 pt-4 shrink-0">
          {tracks.map((t) => (
            <div key={t.id} className="h-10 border-b border-zinc-800 px-3 flex items-center text-xs font-medium text-zinc-300">
              {t.name}
            </div>
          ))}
        </div>
        <div
          onClick={(e) => {
            const rect = e.currentTarget.getBoundingClientRect();
            seekTo((e.clientX - rect.left) / zoomLevel);
          }}
          className="flex-1 relative overflow-x-auto cursor-pointer bg-zinc-950"
        >
          <div className="relative">
            {tracks.map((t) => (
              <div key={t.id} className="h-10 border-b border-zinc-900 relative">
                {clips
                  .filter((c) => c.trackId === t.id)
                  .map((c) => (
                    <div
                      key={c.id}
                      style={{ left: `${c.timelineStart * zoomLevel}px`, width: `${c.duration * zoomLevel}px` }}
                      className={`absolute top-1 bottom-1 rounded border px-2 flex items-center text-[11px] font-semibold text-white ${c.color}`}
                    >
                      <span className="truncate">{c.name}</span>
                    </div>
                  ))}
              </div>
            ))}
          </div>
          <div
            style={{ left: `${currentTime * zoomLevel}px` }}
            className="absolute top-0 bottom-0 w-0.5 bg-rose-500 z-30 pointer-events-none"
          />
        </div>
      </section>

      {isPublishModalOpen && (
        <div className="fixed inset-0 bg-black/80 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className="bg-zinc-900 border border-zinc-700 rounded-2xl max-w-lg w-full p-6 space-y-5 shadow-2xl">
            <div className="flex items-center justify-between">
              <h2 className="text-base font-bold text-white">🚀 Publicación Directa Multicanal</h2>
              <button onClick={() => setIsPublishModalOpen(false)} className="text-zinc-400 hover:text-white">✕</button>
            </div>
            <div className="grid grid-cols-3 gap-2">
              {['tiktok', 'instagram', 'youtube'].map((net) => (
                <button
                  key={net}
                  onClick={() => togglePlatform(net as SocialPlatform)}
                  className={`p-3 rounded-xl border text-left ${
                    selectedPlatforms.includes(net as SocialPlatform)
                      ? 'bg-violet-600/20 border-violet-500 text-white'
                      : 'bg-zinc-950 border-zinc-800 text-zinc-500'
                  }`}
                >
                  <div className="text-xs font-bold uppercase">{net}</div>
                </button>
              ))}
            </div>
            <textarea
              value={postCaption}
              onChange={(e) => setPostCaption(e.target.value)}
              rows={3}
              className="w-full rounded-lg bg-zinc-950 border border-zinc-800 p-2.5 text-xs text-white"
            />
            {publishResults && (
              <div className="p-3 rounded-xl bg-emerald-950/50 border border-emerald-700 text-xs text-emerald-300">
                ✅ Publicado exitosamente en canales seleccionados.
              </div>
            )}
            <div className="flex justify-end gap-2">
              <button onClick={() => setIsPublishModalOpen(false)} className="px-4 py-2 text-xs bg-zinc-800 rounded-lg">Cerrar</button>
              <button
                onClick={handlePublishToSocials}
                disabled={isPublishing}
                className="px-5 py-2 text-xs font-bold bg-violet-600 hover:bg-violet-500 text-white rounded-lg"
              >
                {isPublishing ? 'Publicando...' : '⚡ Publicar Ahora'}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
